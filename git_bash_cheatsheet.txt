# ============================================
# НАСТРОЙКА GIT (один раз при установке)
# ============================================
git config --global user.name "Ваше Имя"          # задать имя пользователя
git config --global user.email "email@mail.com"   # задать email
git config --global --list                        # посмотреть настройки

# ============================================
# СОЗДАНИЕ И КЛОНИРОВАНИЕ РЕПОЗИТОРИЯ
# ============================================
git init                       # создать новый репозиторий в текущей папке
git clone <url>                # склонировать удалённый репозиторий
git clone <url> <папка>        # склонировать в указанную папку

# ============================================
# ОСНОВНЫЕ КОМАНДЫ GIT BASH (навигация по файлам)
# ============================================
pwd                            # показать текущую папку
ls                             # список файлов в папке
ls -la                         # список всех файлов (включая скрытые)
cd <папка>                     # перейти в папку
cd ..                          # вернуться на уровень выше
cd ~                           # перейти в домашнюю папку
mkdir <имя>                    # создать папку
touch <файл>                   # создать пустой файл
rm <файл>                      # удалить файл
rm -r <папка>                  # удалить папку рекурсивно
clear                          # очистить экран терминала

# ============================================
# ПРОВЕРКА СОСТОЯНИЯ
# ============================================
git status                     # показать состояние файлов
git log                        # история коммитов
git log --oneline              # краткая история коммитов
git diff                       # показать несохранённые изменения

# ============================================
# ДОБАВЛЕНИЕ ИЗМЕНЕНИЙ (STAGING)
# ============================================
git add <файл>                 # добавить конкретный файл
git add .                      # добавить все изменения
git add -A                     # добавить все (включая удаления)

# ============================================
# КОММИТЫ
# ============================================
git commit -m "сообщение"      # закоммитить с комментарием
git commit -am "сообщение"     # add + commit для отслеживаемых файлов

# ============================================
# ВЕТКИ (BRANCHES)
# ============================================
git branch                     # список веток
git branch <имя>               # создать ветку
git checkout <имя>             # переключиться на ветку
git checkout -b <имя>          # создать и переключиться на ветку
git switch <имя>               # переключиться (новая команда)
git switch -c <имя>            # создать и переключиться (новая команда)
git merge <ветка>              # слить ветку в текущую
git branch -d <имя>            # удалить ветку

# ============================================
# УДАЛЁННЫЙ РЕПОЗИТОРИЙ (REMOTE)
# ============================================
git remote add origin <url>    # привязать удалённый репозиторий
git remote -v                  # посмотреть удалённые репозитории
git push -u origin main        # первый push ветки main
git push                       # отправить изменения
git pull                       # забрать изменения с сервера
git fetch                      # получить изменения без слияния

# ============================================
# ОТМЕНА ИЗМЕНЕНИЙ
# ============================================
git restore <файл>             # отменить изменения в файле
git restore --staged <файл>    # убрать файл из staging
git reset --soft HEAD~1        # отменить последний коммит, сохранив изменения
git reset --hard HEAD~1        # отменить коммит и удалить изменения (ОСТОРОЖНО!)

# ============================================
# ПОЛЕЗНЫЕ СОКРАЩЕНИЯ
# ============================================
# Ctrl+C — прервать команду
# Стрелка вверх/вниз — история команд
# Tab — автодополнение
# q — выйти из режима просмотра (git log)
Основной рабочий процесс:
git status                     # 1. Проверить, что изменилось
git add .                      # 2. Добавить все изменения
git commit -m "Описание"       # 3. Закоммитить
git push                       # 4. Отправить на GitHub
